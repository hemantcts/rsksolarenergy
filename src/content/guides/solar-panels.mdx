export const panelType = (p) => {
  const text = [p.title, p.summary, ...p.features, ...(p.specs?.rows ?? []).flatMap((r) => ('values' in r ? [r.label, ...r.values] : []))].join(' ');
  const bifacial = /bi-?facial/i.test(text) ? ' bifacial' : '';
  if (/topcon/i.test(text)) return `N-type TOPCon${bifacial}`;
  if (/mono|perc/i.test(text)) return `Mono PERC${bifacial}`;
  if (/poly/i.test(text)) return 'Polycrystalline';
  return 'See specification';
};

export const panelWatts = (p) => parseInt(p.title.match(/(\d{2,3})\s*(?:-?\s*watts?\b|wp?\b)/i)?.[1] ?? p.slug.match(/(?:^|-)(\d{2,3})(?:w|wp)(?:-|$)/)?.[1] ?? '0', 10);

export const panelDcr = (p) => {
  const text = [p.slug, p.title, p.summary, ...p.features, ...(p.specs?.rows ?? []).flatMap((r) => ('values' in r ? [r.label, ...r.values] : []))].join(' ');
  if (/\bndcr\b|non[- ]?dcr/i.test(text)) return 'Non-DCR';
  if (/\bdcr\b|domestic content/i.test(text)) return 'DCR';
  return 'Ask us';
};

## Reading a panel datasheet

- Maximum power (Pmax) is the rating in watts at standard test conditions: 1000 W/m² of sunlight at a cell temperature of 25 °C. Punjab rooftops in May run far hotter than that, so real midday output is lower, and our calculator allows for it.
- Module efficiency is the share of sunlight turned into electricity. Higher efficiency means more watts per square foot of roof, which matters when the roof is small.
- Voc and Isc are the open-circuit voltage and short-circuit current. The installer uses them to match the panel strings to the inverter.
- A positive power tolerance means the panel won't produce less than its rating at test conditions.
- UTL lists a 25-year performance warranty on its panels, and the product warranty for defects is separate. Check both on the datasheet for the model you buy.

## The check that decides the subsidy

PM Surya Ghar only pays for systems built with DCR modules from the ALMM list, meaning panels made in India with Indian-made cells. The inspector checks the model, and if it isn't on the list the subsidy is lost and can't be recovered after installation. Ask for the exact model number on your quote and check it against the current ALMM list before the panels go up.

## Panels alone, or a complete system?

If you are replacing panels or adding to an existing system, match the voltage and current to your inverter or charge controller. If you are starting from scratch, buy a matched system, with the panels, inverter, mounting structure and wiring designed together. That avoids the mismatches that waste generation.
